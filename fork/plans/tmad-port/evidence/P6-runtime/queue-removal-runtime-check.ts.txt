import { createHash } from "node:crypto";
import { readFile, rm } from "node:fs/promises";
import { createTestPaseoDaemon } from "./test-utils/paseo-daemon.js";
import { createTestAgentClient } from "./test-utils/fake-agent-client.js";
import { DaemonClient } from "./test-utils/daemon-client.js";

async function main() {
  const homeRoot = process.env.TMAD_AUDIT_HOME!;
  const queuePath = `${homeRoot}/.paseo/queues/12345678-1234-4123-8123-123456789abc.json`;
  const sha = (value: Buffer) => createHash("sha256").update(value).digest("hex");
  const before = sha(await readFile(queuePath));
  let providerTurns = 0;
  let daemon: Awaited<ReturnType<typeof createTestPaseoDaemon>> | undefined;
  let client: DaemonClient | undefined;
  try {
    daemon = await createTestPaseoDaemon({
      paseoHomeRoot: homeRoot,
      cleanup: false,
      mcpEnabled: false,
      agentClients: {
        codex: createTestAgentClient("codex", { onStartTurn: () => { providerTurns += 1; } }),
      },
    });
    client = new DaemonClient({
      url: `ws://127.0.0.1:${daemon.port}/ws`,
      appVersion: "0.1.82",
      reconnect: { enabled: false },
    });
    await client.connect();
    const agent = await client.createAgent({
      agentId: "12345678-1234-4123-8123-123456789abc",
      provider: "codex",
      cwd: "/tmp",
      title: "Copied pending queue runtime proof",
    });
    await new Promise((resolve) => setTimeout(resolve, 1500));
    const live = await readFile(queuePath);
    const afterLive = sha(live);
    await client.close();
    client = undefined;
    const staticDir = daemon.staticDir;
    await daemon.close();
    daemon = undefined;
    await rm(staticDir, { recursive: true, force: true });
    const afterStop = await readFile(queuePath);
    const after = sha(afterStop);
    if (before !== afterLive || before !== after) {
      throw new Error("queue bytes changed in queue-free runtime");
    }
    if (providerTurns !== 0) throw new Error(`fake provider received ${providerTurns} queued turns`);
    const parsed = JSON.parse(afterStop.toString("utf8"));
    if (
      parsed.agentId !== agent.id ||
      parsed.items?.length !== 1 ||
      parsed.items[0]?.id !== "pending-copy-1" ||
      parsed.items[0]?.deliveryState !== "pending"
    ) {
      throw new Error("copied pending item did not remain intact");
    }
    console.log(
      JSON.stringify(
        {
          phase: "queue-free-runtime",
          isolatedHome: homeRoot,
          queuePath,
          queueSha256Before: before,
          queueSha256WhileRuntimeActive: afterLive,
          queueSha256AfterStop: after,
          pendingIds: parsed.items.map((item: { id: string }) => item.id),
          fakeProviderQueuedTurns: providerTurns,
          daemonPortWasEphemeralLoopback: true,
          agentId: agent.id,
        },
        null,
        2,
      ),
    );
  } finally {
    await client?.close().catch(() => undefined);
    if (daemon) {
      const staticDir = daemon.staticDir;
      await daemon.close().catch(() => undefined);
      await rm(staticDir, { recursive: true, force: true });
    }
  }
}

await main();
